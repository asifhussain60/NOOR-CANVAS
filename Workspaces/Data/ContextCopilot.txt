    <!-- Script to render Lucide icons and handle view switching -->
    <script>
        lucide.createIcons();

        const userView = document.getElementById('user-view-container');
        const hostView = document.getElementById('host-view-container');
        const userBtn = document.getElementById('user-btn');
        const hostBtn = document.getElementById('host-btn');

        /**
         * Switches the active view and updates the button styles.
         * @param {string} viewId - The ID of the view container to show ('user-view-container' or 'host-view-container').
         */
        function showView(viewId) {
            // Hide all views
            userView.style.display = 'none';
            hostView.style.display = 'none';

            // Deactivate all buttons
            userBtn.classList.remove('active');
            hostBtn.classList.remove('active');

            // Show the selected view and activate the corresponding button
            if (viewId === 'user-view-container') {
                userView.style.display = 'block';
                userBtn.classList.add('active');
            } else if (viewId === 'host-view-container') {
                hostView.style.display = 'block';
                hostBtn.classList.add('active');
            }
        }

        // Add event listeners
        userBtn.addEventListener('click', () => showView('user-view-container'));
        hostBtn.addEventListener('click', () => showView('host-view-container'));

        // Initialize view on load (default to USER VIEW)
        window.onload = function() {
            showView('user-view-container');
            lucide.createIcons();
        };

    </script>
